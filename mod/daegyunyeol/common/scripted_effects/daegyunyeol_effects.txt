# 여러 곳에서 쓰거나, 나중에 값을 고쳐야 하는 효과는 여기 모아둔다.
# 파일 하나만 고치면 포커스와 이벤트 양쪽에 반영된다.

#####################################################################
# 대한민국 초기 설정
#
# 어떤 경로로 KOR 이 생기든(일본 디시전, 바닐라 해방, 콘솔 release) 한 번만 실행된다.
# on_daily_KOR 과 daegyunyeol.1 양쪽에서 호출하므로 안쪽 플래그로 중복을 막는다.
#
# 포커스트리를 여기서 명시적으로 불러온다. country = { } 점수만 믿으면
# 게임 시작 시점에 존재하지 않던 나라에서 제네릭 트리가 나올 수 있다.
#####################################################################

KOR_daegyunyeol_setup = {
	if = {
		limit = { NOT = { has_country_flag = KOR_daegyunyeol_setup_done } }

		set_country_flag = KOR_daegyunyeol_setup_done

		load_focus_tree = kor_daegyunyeol_tree

		# 정치 노선 포커스가 promote_character 로 승격시키려면 먼저 영입되어 있어야 한다.
		recruit_character = KOR_leader_democratic
		recruit_character = KOR_leader_authoritarian

		add_ideas = KOR_rift_demilitarization
		KOR_disband_all_units = yes
	}
}

#####################################################################
# 군대 해산
#
# HOI4 에는 "모든 사단 해산" 한 줄짜리 효과가 없다. 사단 템플릿 이름을
# 하나씩 지정해서 delete_unit_template_and_units 로 지워야 한다.
# disband = yes 면 장비와 인력이 회수되고, no 면 그냥 사라진다.
#
# 아래 이름은 KOR 이 가질 법한 바닐라 템플릿 추정치다. 이름이 실제와
# 다르면 error.log 에 한 줄 남고 그냥 넘어간다(크래시하지 않는다).
# 게임을 한 번 돌린 뒤 error.log 를 보고 정확한 이름으로 고칠 것.
# 해방 직후 KOR 에 사단이 아예 없으면 이 효과는 아무 일도 하지 않는다.
#####################################################################

KOR_disband_all_units = {
	delete_unit_template_and_units = {
		division_template = "Infantry Division"
		disband = yes
	}
	delete_unit_template_and_units = {
		division_template = "Cavalry Division"
		disband = yes
	}
	delete_unit_template_and_units = {
		division_template = "Garrison Brigade"
		disband = yes
	}
	delete_unit_template_and_units = {
		division_template = "Hohei Shidan"
		disband = yes
	}
	delete_unit_template_and_units = {
		division_template = "Kihei Ryodan"
		disband = yes
	}
}

#####################################################################
# 무장 금지 해제
#
# '군수 전환' 을 건너뛰고 '균열군 창설' 로 직행했을 수 있으므로
# 두 단계 중 어느 쪽을 달고 있든 처리한다.
#####################################################################

KOR_lift_arms_ban = {
	if = {
		limit = { has_idea = KOR_rift_demilitarization }
		remove_ideas = KOR_rift_demilitarization
	}
	if = {
		limit = { has_idea = KOR_limited_rearmament }
		remove_ideas = KOR_limited_rearmament
	}
}
