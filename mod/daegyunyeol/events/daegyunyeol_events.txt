add_namespace = daegyunyeol

# 1. 일본 - 대균열이 열리다
country_event = {
	id = daegyunyeol.1
	title = daegyunyeol.1.t
	desc = daegyunyeol.1.d

	is_triggered_only = yes

	option = {
		name = daegyunyeol.1.a

		release = KOR

		KOR = {
			load_focus_tree = kor_daegyunyeol_tree
			# 포커스에서 promote_character 로 승격시키려면 먼저 영입되어 있어야 한다.
			recruit_character = KOR_leader_democratic
			recruit_character = KOR_leader_authoritarian
			# 균열 직후의 한반도에는 군대를 만들 기반이 없다.
			# 니케 '군수 전환' 으로 완화되고 '균열군 창설' 로 해제된다.
			add_ideas = KOR_rift_demilitarization
			add_political_power = 150
			country_event = { id = daegyunyeol.2 hours = 4 }
		}
	}
}
